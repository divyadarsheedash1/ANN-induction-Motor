import torch
from torch import nn

class MLP(nn.Module):
    
    def __init__(self):
        super().__init__()
        
        self.layer_stack = nn.Sequential(
            
            nn.Linear(4, 64),
            nn.ReLU(),
            nn.Dropout(0.2),
            
            nn.Linear(64, 64),
            nn.ReLU(),
            nn.Dropout(0.2),
            
            nn.Linear(64, 6)
        )
        
    def forward(self, x):
        
        return self.layer_stack(x)
